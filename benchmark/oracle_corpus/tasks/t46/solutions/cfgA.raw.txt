def prune(tree, max_depth):
    def _prune(node, current_depth):
        if current_depth > max_depth:
            return None
        
        new_children = []
        for child in node.get('children', []):
            pruned_child = _prune(child, current_depth + 1)
            if pruned_child is not None:
                new_children.append(pruned_child)
        
        if node['kind'] == 'dir' and not new_children and current_depth != 0:
            return None
        
        return {
            "name": node["name"],
            "kind": node["kind"],
            "children": new_children
        }
    
    return _prune(tree, 0)
