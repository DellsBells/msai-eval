def neighbor_sum_grid(grid):
    rows = len(grid)
    cols = len(grid[0])
    
    result = [[0] * cols for _ in range(rows)]
    
    for i in range(1, rows - 1):
        for j in range(1, cols - 1):
            result[i][j] = sum(grid[x][y] for x in (i-1, i+1) for y in (j-1, j+1))
    
    # Handle the edges
    for i in range(rows):
        for j in range(cols):
            if i == 0:
                result[i][j] = sum(grid[x][y] for x in [i, i + 1] for y in [-1, 0, 1])
            elif i == rows - 1:
                result[i][j] = sum(grid[x][y] for x in [i-1, i] for y in [-1, 0, 1])
            else:
                result[i][j] = grid[i][j]
    
    # Handle the corners
    if cols > 2 and rows > 2:
        for j in range(cols):
            for i in (0, -1, -rows + 1):
                result[i][j] = sum(grid[x][y] for x in [-1, 0, 1] for y in [i-1, i, i+1])
    
    return result
