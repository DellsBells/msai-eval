def block_similarity(a: str, b: str) -> float:
    def longest_matching_block(s1, s2):
        max_len = 0
        max_i = 0
        max_j = 0
        len_s1 = len(s1)
        len_s2 = len(s2)
        dp = [[0] * (len_s2 + 1) for _ in range(len_s1 + 1)]
        
        for i in range(1, len_s1 + 1):
            for j in range(1, len_s2 + 1):
                if s1[i - 1] == s2[j - 1]:
                    dp[i][j] = dp[i - 1][j - 1] + 1
                    if dp[i][j] > max_len:
                        max_len = dp[i][j]
                        max_i = i - max_len
                        max_j = j - max_len
        
        return max_len, max_i, max_j

    def M(s1, s2):
        if not s1 or not s2:
            return 0
        k, i, j = longest_matching_block(s1, s2)
        if k == 0:
            return 0
        return k + M(s1[:i], s2[:j]) + M(s1[i + k:], s2[j + k:])

    if not a and not b:
        return 1.0
    total_matched = M(a, b)
    return round(2 * total_matched / (len(a) + len(b)), 4)
