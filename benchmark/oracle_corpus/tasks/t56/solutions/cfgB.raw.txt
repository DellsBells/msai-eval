def dispatch(num_elevators, calls):
    # Simulate an elevator bank dispatch algorithm.
    # Each call is represented by a tuple (floor, direction), where:
    # - floor: the current or target floor of the call.
    # - direction: 1 if going up and -1 if going down.
    
    # Initialize elevators with zero floors
    elevators = [0] * num_elevators
    
    # Process each call in calls
    for call_floor, call_direction in calls:
        min_distance = float('inf')
        selected_elevator_index = None
        
        # Find the elevator that is closest to the current floor of the call.
        for i in range(num_elevators):
            distance_to_call = abs(call_floor - elevators[i])
            if distance_to_call < min_distance:
                min_distance = distance_to_call
                selected_elevator_index = i
                
        # Update the selected elevator's position based on direction and floor of the call.
        new_position = elevators[selected_elevator_index] + (call_direction * abs(call_floor))
        elevators[selected_elevator_index] = new_position
        
    return elevators

# Example usage
dispatch(1, [(3, 0)]) # Output: [3]
